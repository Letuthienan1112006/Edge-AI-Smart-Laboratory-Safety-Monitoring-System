# Edge device. Keep this list short: every package installed here is a
# package that can show up in a power figure.
#
#   sudo apt install -y python3-opencv python3-numpy python3-yaml \
#                       python3-psutil python3-libgpiod gpiod
#   pip install --break-system-packages -r requirements/edge.txt
#
# Do NOT install torch or tensorflow on the board. The NCNN / tflite-runtime
# / RKNN paths exist precisely so the device never carries a training framework.

# --- CPU runtimes (any aarch64 board) ---
ncnn>=1.0.2024
tflite-runtime>=2.14 ; platform_machine == "aarch64"
onnxruntime>=1.17

# --- Rockchip RK3588/RK3588S NPU (Orange Pi 5) ---
# On-device runtime only. Conversion ONNX -> .rknn happens on an x86 host
# with `rknn-toolkit2`, which is a DIFFERENT package. Both must match the
# librknnrt.so shipped in the board image, or you get errors that do not
# name the version as the cause.
rknn-toolkit-lite2 ; platform_machine == "aarch64"

# --- GPIO ---
# libgpiod is the kernel's own interface and works on any board with
# /dev/gpiochip*. Prefer the apt package (python3-libgpiod) over pip.
# The board-specific libraries below are fallbacks the code also accepts:
#   RPi.GPIO   -- Raspberry Pi only
#   OPi.GPIO   -- Orange Pi
